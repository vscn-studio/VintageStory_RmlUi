cmake_minimum_required(VERSION 3.24)
project(vsrmlui_native VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(FT_DISABLE_ZLIB ON CACHE BOOL "" FORCE)
set(FT_DISABLE_BZIP2 ON CACHE BOOL "" FORCE)
set(FT_DISABLE_PNG ON CACHE BOOL "" FORCE)
set(FT_DISABLE_HARFBUZZ ON CACHE BOOL "" FORCE)
set(FT_DISABLE_BROTLI ON CACHE BOOL "" FORCE)
FetchContent_Declare(freetype URL https://codeload.github.com/freetype/freetype/tar.gz/refs/tags/VER-2-13-3 URL_HASH SHA256=bc5c898e4756d373e0d991bab053036c5eb2aa7c0d5c67e8662ddc6da40c4103 TIMEOUT 120)
FetchContent_MakeAvailable(freetype)
if(NOT TARGET Freetype::Freetype)
  add_library(Freetype::Freetype ALIAS freetype)
endif()
set(RMLUI_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../RmlUi" CACHE PATH "Pinned RmlUi source checkout")
set(RMLUI_SAMPLES OFF CACHE BOOL "" FORCE)
set(RMLUI_TESTS OFF CACHE BOOL "" FORCE)
set(RMLUI_LUA_BINDINGS OFF CACHE BOOL "" FORCE)
set(RMLUI_SVG_PLUGIN OFF CACHE BOOL "" FORCE)
set(RMLUI_LOTTIE_PLUGIN OFF CACHE BOOL "" FORCE)
add_subdirectory("${RMLUI_SOURCE_DIR}" "rmlui" EXCLUDE_FROM_ALL)
# Internal select values use the content area, leaving arrow padding clear.
file(READ "${RMLUI_SOURCE_DIR}/Source/Core/Elements/WidgetDropDown.cpp" dropdown)
set(value_layout "ElementUtilities::FormatElement(value_element, parent_element->GetBox().GetSize(BoxArea::Border));")
string(FIND "${dropdown}" "${value_layout}" found_value_layout)
if(found_value_layout EQUAL -1)
  message(FATAL_ERROR "RmlUi dropdown changed; review selected-value layout.")
endif()
string(REPLACE "${value_layout}" "ElementUtilities::FormatElement(value_element, parent_element->GetBox().GetSize(BoxArea::Content));" dropdown "${dropdown}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/WidgetDropDown.cpp" "${dropdown}")
get_target_property(core_sources rmlui_core SOURCES)
list(FILTER core_sources EXCLUDE REGEX "WidgetDropDown\\.cpp$")
set_property(TARGET rmlui_core PROPERTY SOURCES "${core_sources};${CMAKE_CURRENT_BINARY_DIR}/WidgetDropDown.cpp")
target_include_directories(rmlui_core PRIVATE "${RMLUI_SOURCE_DIR}/Source/Core/Elements")
# Resolve an OS fallback only when the default font engine encounters a missing glyph.
file(READ "${RMLUI_SOURCE_DIR}/Source/Core/FontEngineDefault/FontFaceHandleDefault.cpp" font_face)
set(font_needle "else if (look_in_fallback_fonts)\n\t\t{\n\t\t\tconst int num_fallback_faces")
string(FIND "${font_face}" "${font_needle}" found_font_hook)
if(found_font_hook EQUAL -1)
  message(FATAL_ERROR "RmlUi font fallback changed; review system font integration.")
endif()
string(REPLACE "${font_needle}" "else if (look_in_fallback_fonts)\n\t\t{\n\t\t\tvsrml_resolve_font(static_cast<uint32_t>(character));\n\t\t\tconst int num_fallback_faces" font_face "${font_face}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/FontFaceHandleDefault.cpp" "#include <cstdint>\nextern void vsrml_resolve_font(uint32_t character);\n${font_face}")
get_target_property(core_sources rmlui_core SOURCES)
list(FILTER core_sources EXCLUDE REGEX "FontFaceHandleDefault\\.cpp$")
set_property(TARGET rmlui_core PROPERTY SOURCES "${core_sources};${CMAKE_CURRENT_BINARY_DIR}/FontFaceHandleDefault.cpp")
target_include_directories(rmlui_core PRIVATE "${RMLUI_SOURCE_DIR}/Source/Core/FontEngineDefault")
# Patch only the generated build copy. Never modify the upstream checkout.
file(READ "${RMLUI_SOURCE_DIR}/Backends/RmlUi_Renderer_GL3.cpp" renderer)
set(needle "// Draw to backbuffer\n\tglBindFramebuffer(GL_FRAMEBUFFER, 0);")
string(FIND "${renderer}" "${needle}" found)
if(found EQUAL -1)
  message(FATAL_ERROR "RmlUi GL3 backend changed; review framebuffer integration before upgrading.")
endif()
string(REPLACE "${needle}" "// Composite into the host game's render target.\n\tglBindFramebuffer(GL_FRAMEBUFFER, vsrml_target_framebuffer());" renderer "${renderer}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/Renderer.cpp" "extern unsigned int vsrml_target_framebuffer();\n${renderer}")
add_library(vsrmlui_native SHARED bridge.cpp "${CMAKE_CURRENT_BINARY_DIR}/Renderer.cpp")
target_include_directories(vsrmlui_native PRIVATE "${RMLUI_SOURCE_DIR}/Backends")
target_link_libraries(vsrmlui_native PRIVATE RmlUi::Core ${CMAKE_DL_LIBS})
if(MSVC)
  target_compile_options(vsrmlui_native PRIVATE /utf-8 /W4)
endif()
# A fixed path for both single-config Ninja/Make and multi-config Visual Studio.
set_target_properties(vsrmlui_native PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/out/$<CONFIG>"
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/out/$<CONFIG>"
  CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(APPLE)
  set_target_properties(vsrmlui_native PROPERTIES MACOSX_RPATH ON INSTALL_NAME_DIR "@rpath")
endif()
install(TARGETS vsrmlui_native RUNTIME DESTINATION . LIBRARY DESTINATION .)
